import os
from openpyxl import Workbook

OUT = "uploads"
os.makedirs(OUT, exist_ok=True)


def build(def_code, sheet_name, field_labels, field_codes, rows, path):
    """Mirror ExcelTemplateBuilder/ExcelWriter layout: data sheet + hidden _meta sheet."""
    wb = Workbook()
    ws = wb.active
    ws.title = sheet_name
    ws.append(field_labels)
    for r in rows:
        ws.append(r)
    meta = wb.create_sheet("_meta")
    meta.append(["def_code"] + field_labels)
    meta.append([def_code] + field_codes)
    meta.sheet_state = "hidden"
    wb.save(path)
    print("wrote", path, "rows=", len(rows))
